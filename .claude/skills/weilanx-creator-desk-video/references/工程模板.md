# 工程模板

最小可运行例子：`src/library/previews/WarmDeskShowDemo.tsx`（Studio 里 `library-warm-desk-show`，无声，5 个场景）；真人出镜的完整样片：`src/episodes/ep26-opus-video/`。下面是一期正式口播的文件清单和模板，最后一节是真人出镜版要多加的东西。

```text
src/episodes/<期>/
├── cues.json        npm run cues 生成
├── cues.ts          makeCues
├── scenes.tsx       DeskScene[] 场景表（场景多可以拆 scenes/part1.tsx、part2.tsx）
├── Cover.tsx        竖版封面（DeskCoverTall）+ 横版首帧封面（DeskCoverWide）
└── Ep<NN><Name>.tsx DeskShowConfig + 组件
audio/episodes/<期>.mjs
tests/episodes/<期>.test.ts
```

## cues.ts

```ts
import raw from './cues.json';
import {makeCues} from '../../library/cues';

export const VOICE_SECONDS = 142.6; // 人声时长 + 0.3～0.5 秒，且不短于最后一条字幕结束
export const {CUES, DURATION, cf, ce, cw, cueAt} = makeCues(raw, VOICE_SECONDS);
```

## tests/episodes/&lt;期&gt;.test.ts

```ts
import {existsSync} from 'node:fs';
import {describe, expect, it} from 'vitest';
import {CUES, DURATION} from '../../src/episodes/<期>/cues';
import {SCENES, STEPS} from '../../src/episodes/<期>/scenes';

describe('<期>', () => {
  it('场景按时间递增，都在片长内', () => {
    for (let i = 1; i < SCENES.length; i++) expect(SCENES[i].from).toBeGreaterThan(SCENES[i - 1].from);
    SCENES.forEach((s) => expect(s.from).toBeLessThan(DURATION));
  });
  it('步骤号不越界，正文覆盖了每一步', () => {
    const used = new Set(SCENES.filter((s) => s.step !== undefined).map((s) => s.step));
    used.forEach((k) => expect(k).toBeLessThan(STEPS.length));
    STEPS.forEach((_, k) => expect(used.has(k)).toBe(true));
  });
  it('字幕都在片长内', () => {
    CUES.forEach((c) => expect(c.end * 30).toBeLessThanOrEqual(DURATION));
  });
  it('音效文件存在', () => {
    SCENES.flatMap((s) => s.sfx).forEach(([name, at]) => {
      expect(at).toBeLessThan(DURATION);
      expect(existsSync(name.includes('/') ? `public/${name}` : `public/audio/sfx/${name}.wav`)).toBe(true);
    });
  });
});
```

（`cw(n, '词')` 找不到词会在模块加载时直接抛错，所以「改了字幕跑一遍测试」就能发现挂错的词。）

## 真人出镜版（用户给了自己录好的口播视频）

样片 `src/episodes/ep26-opus-video/`。在上面的基础上多做这几件事。

### 人声和出镜视频

```sh
SRC=<录制原件>.MOV
# 人声：先测响度，增益调到 −19 LUFS（ep26：原片 −22.7 LUFS → +3.6 dB）
ffmpeg -nostats -i "$SRC" -vn -af ebur128 -f null - 2>&1 | grep -E '^\s+I:'
ffmpeg -y -i "$SRC" -vn -af "volume=<增益>dB,alimiter=limit=0.8:level=false" -ar 48000 -ac 2 -c:a pcm_s16le public/audio/<期>/voice.wav
# 出镜视频：居中裁方形、720×720、30fps、H.264、去音轨（小窗静音，人声只走 voice.wav）
ffmpeg -y -i "$SRC" -an -vf "crop='min(iw,ih)':'min(iw,ih)',scale=720:720,fps=30" -c:v libx264 -crf 20 -preset medium -pix_fmt yuv420p -movflags +faststart public/footage/<期>/host.mp4
```

人物不在画面正中时，把 `crop` 的 x / y 偏移写上（`crop=w:h:x:y`），保证脸在方框上三分之一。

### Ep&lt;NN&gt;&lt;Name&gt;.tsx：出镜小窗挂 overlay

```tsx
import {clamp01, easeInOut} from '../../library/anim';
import {HostCam} from '../../library/warm-desk/HostCam';

/** A-roll 大（人是主角）、B-roll 小（卡片是主角）；左下角对齐，底边都在 y=920（字幕在下面） */
const POSE = {A: {x: 60, size: 540}, B: {x: 46, size: 330}};
const BOTTOM = 920;
const TURN = 12;

/** 第 f 帧小窗的边长和位置：每个场景开始时 12 帧内从上一个姿态过渡到这个姿态 */
export const camAt = (f: number) => {
  let prev = POSE[SCENES[0].roll];
  let cur = prev;
  let from = 0;
  for (let i = 0; i < SCENES.length && SCENES[i].from <= f; i++) {
    prev = cur;
    cur = POSE[SCENES[i].roll];
    from = SCENES[i].from;
  }
  const p = easeInOut(clamp01((f - from) / TURN));
  const size = prev.size + (cur.size - prev.size) * p;
  return {x: prev.x + (cur.x - prev.x) * p, y: BOTTOM - size, size};
};

const Cam: React.FC<{f: number}> = ({f}) => {
  const c = camAt(f);
  return <HostCam src="footage/<期>/host.mp4" x={c.x} y={c.y} size={c.size} radius={16} name="蔚蓝" role={c.size > 450 ? 'AI 实战分享' : undefined} f={f} />;
};

const CONFIG: DeskShowConfig = {
  // …同上
  overlay: Cam,
  mascot: {x: 1700, y: 1000, s: 0.75},
  decor: [[0, 340, 640, 620], [440, 120, 1040, 780]], // 避开小窗和主内容区
};
```

import {camAt} from '../../src/episodes/<期>/Ep<NN><Name>';

it('出镜小窗始终在左下角：左边 ≤ 60、底边 = 920，A-roll 大、B-roll 小', () => {
  for (let f = 0; f < DURATION; f += 15) {
    const c = camAt(f);
    expect(c.x).toBeLessThanOrEqual(60);
    expect(Math.round(c.y + c.size)).toBe(920);
  }
  const a = SCENES.find((s) => s.roll === 'A')!;
  const b = SCENES.find((s) => s.roll === 'B')!;
  expect(camAt(a.from + 20).size).toBeGreaterThan(camAt(b.from + 20).size);
});
